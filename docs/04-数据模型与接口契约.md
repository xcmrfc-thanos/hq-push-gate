## 1. Protobuf 消息定义（内部链路）

> 权威契约以 `flink/src/main/proto/hq/v1/*.proto`（buf 管理、随契约冻结变更走 ADR）为准，本节为文档节选——与代码冲突时以 proto 为准。

```protobuf
syntax = "proto3";
package hq.v1; // 注：文档节选笔误已修正（proto 实际为 hq.v1，2026-09-07 Step1 D04-1）

enum Market { A_SHARE = 0; HK = 1; US = 2; CRYPTO = 3; }

// 标准化行情 tick —— 全链路统一携带 market
message Tick {
  Market market       = 1;
  string symbol       = 2;   // 600000 / AAPL / 00700 / BTC-USDT
  string exchange     = 3;
  int64  timestamp_ms = 4;
  double last_price   = 5;
  double open         = 6;
  double high         = 7;
  double low          = 8;
  double pre_close    = 9;   // 昨收，用于涨跌幅
  double volume       = 10;  // 累计成交量
  double amount       = 11;  // 累计成交额
}

message Kline {                       // 周期闭合K线
  Market market       = 1;
  string symbol       = 2;
  int32  period_min   = 3;   // 1/5/15/30/60/1440(日)
  int64  begin_ts_ms  = 4;
  double open = 5; double high = 6; double low = 7; double close = 8;
  double volume = 9; double amount = 10;
}

// 规则广播（Flink broadcast stream 消费）
message RuleMsg {
  int64  rule_id      = 1;
  int64  version      = 2;   // 单调递增，幂等去重
  string op           = 3;   // UPSERT / DELETE
  Market market       = 4;
  string symbol       = 5;
  RuleType type       = 6;   // PRICE_ABOVE/PRICE_BELOW/PRICE_RANGE/PCT_CHANGE/INDICATOR
  string condition    = 7;   // 条件 JSON（schema v2 见 ADR-042/docs/10 §11）：{"threshold":12.5} 或
                             //   {"indicator":"VOLUME","op":">","value":1e6}；公共叠加字段 once/session/expire_at；
                             //   PCT_CHANGE 规范形 = 正幅值 + direction(up/down/both)；组合(all/any)仅查询侧
  int32  cooldown_sec = 8;   // 防抖冷却
}

// 条件 schema v2 要点（ADR-042，权威全量定义 = docs/10 §11 + tests/fixtures/conditions.v2.json）：
// 1. 数值一律有界：|threshold| ≤ 1e6(价)/100(pct)/1e12(量)，恒为正幅值（负值输入由解析/校验层归一化）；
// 2. 查询侧条件含 "type" 字段（ai-query condition.py）；规则侧 type 在 RuleMsg，condition 无 "type"；
// 3. 规则侧仅扁平条件（biz RuleConditionValidator 拒绝 all/any），畸形条件 400 拒绝不入库；
// 4. once/session/expire_at 为可选叠加字段，Flink 实时路径当前未强制（登记项）。

// 告警事件（notify 消费）
message AlertEvent {
  string event_id     = 1;   // rule_id + rule_version + trigger_window
  int64  rule_id      = 2;
  int64  version      = 3;
  Market market       = 4;
  string symbol       = 5;
  string title        = 6;
  string detail_json  = 7;   // 触发时快照：价格/指标值/时间
  int64  trigger_ts   = 8;
}

message AlertDelivery {
  string delivery_id  = 1;   // event_id + user_id
  string event_id     = 2;
  int64  user_id      = 3;
  Market market       = 4;
  string symbol       = 5;
  string title        = 6;
  string detail_json  = 7;
  int64  trigger_ts   = 8;
}

message WsPushMsg {                   // 固定 ws_push topic，按 gateway_slot 路由
  string gateway_id = 1;
  uint32 gateway_slot = 2;
  oneof payload { QuoteBatch quotes = 3; AlertDelivery alert = 4; Kline kline = 5; }
}
message QuoteBatch { repeated Tick ticks = 1; }  // 合并窗口内每 symbol 只保留最新
```

```sql
CREATE TABLE alert_rule (

  id              BIGINT PRIMARY KEY,
  aggregate_type  VARCHAR(32) NOT NULL, -- RULE / GROUP / SUBSCRIPTION
  aggregate_id    BIGINT NOT NULL,

  published_at    DATETIME(3) NULL,
  last_error      VARCHAR(512) NULL,

  updated_at      DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3),
  UNIQUE KEY uk_event_key (event_key),

CREATE TABLE alert_record (            -- 用户投递记录（notify 内嵌 alert-inbox 的落库表）
  id         BIGINT PRIMARY KEY,
  delivery_id VARCHAR(96) NOT NULL UNIQUE,  -- event_id + user_id
  event_id   VARCHAR(64) NOT NULL,
  rule_id    BIGINT NOT NULL,
  user_id    BIGINT NOT NULL,
  market     VARCHAR(16) NOT NULL,
  symbol     VARCHAR(32) NOT NULL,
  title      VARCHAR(128),
  trigger_at DATETIME(3) NOT NULL,
  status     VARCHAR(16) NOT NULL DEFAULT 'PENDING', -- PENDING/SENT/ACKED/EXPIRED
  cursor_id  BIGINT NOT NULL,          -- 全局投递序号（单调递增，补拉游标）

CREATE TABLE watch_group_item (

```

### 数据保留与清理（数据生命周期治理，docs/12 §1.1 #19）

| 表 | 策略 | 默认保留期 | 载体 |
|---|---|---|---|
| outbox_event | 仅删 `status='PUBLISHED'` 且 `published_at` 超期的行；PENDING/FAILED 保留供排查/重放 | 7d（`OUTBOX_PUB_RETENTION`） | outbox-publisher 清理 Job：每小时扫描，单批 1000（`OUTBOX_PUB_CLEANUP_*`） |
| notify_send_log | 按 `created_at` 批量删除 | 90d（`NOTIFY_SEND_LOG_RETENTION`） | notify 清理 Job：每小时扫描，单批 1000（`NOTIFY_CLEANUP_*`） |
| alert_record | 不物理删除：超过 72h 补拉保留窗口的行置 `EXPIRED`（补拉按 cursor+窗口判空，旧行不参与投递） | 永久（U2 起按量评估归档） | notify ExpireStale 扫描（docs/07 §6） |
| tick/kline（CK） | TTL 12 个月 + S3 归档设计（docs/00 §3 数据维度） | 见 §4 DDL | ClickHouse TTL |

清理索引：V9 迁移为 outbox_event 增加 `(status, published_at)`、为 notify_send_log 增加 `(created_at)`；删除按批 LIMIT 执行以控制单事务锁窗口，指标 `outbox_cleanup_deleted_total` / `notify_send_log_cleanup_total`。

## 4. ClickHouse DDL（时序明细）

```sql
CREATE TABLE tick_local ON CLUSTER ck_cluster (
  event_id String,
  ingest_version UInt64,
  market LowCardinality(String),
  symbol LowCardinality(String),
  ts_ms  Int64,
  last_price Float64, open Float64, high Float64, low Float64, pre_close Float64,
  volume Float64, amount Float64
) ENGINE = ReplicatedReplacingMergeTree('/clickhouse/tables/{shard}/tick_local', '{replica}', ingest_version)
PARTITION BY (market, toYYYYMM(toDateTime(intDiv(ts_ms,1000))))
ORDER BY (market, symbol, ts_ms, event_id)
TTL toDateTime(intDiv(ts_ms,1000)) + INTERVAL 12 MONTH;

CREATE TABLE tick_distributed ON CLUSTER ck_cluster AS tick_local
ENGINE = Distributed(ck_cluster, default, tick_local, rand());

CREATE TABLE kline_local ON CLUSTER ck_cluster (
  event_id String, ingest_version UInt64,
  market LowCardinality(String), symbol LowCardinality(String),
  period_min UInt16, begin_ts Int64,
  open Float64, high Float64, low Float64, close Float64,
  volume Float64, amount Float64
) ENGINE = ReplicatedReplacingMergeTree('/clickhouse/tables/{shard}/kline_local', '{replica}', ingest_version)
PARTITION BY (market, period_min, toYYYYMM(toDateTime(intDiv(begin_ts,1000))))
ORDER BY (market, symbol, period_min, begin_ts, event_id);

-- 物化视图：1m 聚合 5m（示意）
CREATE MATERIALIZED VIEW mv_kline_5m ... ENGINE = ReplicatedSummingMergeTree ... AS
SELECT ... FROM kline_local WHERE period_min = 1 ...;
```

Doris 侧：`CREATE EXTERNAL TABLE tick_ext ENGINE=ODBC/MySQL PROPERTIES(...) ` 映射 CK 分布式表；自有表 `alert_event_detail`（明细统计）。

## 7. WebSocket 协议（/ws，WSS 握手时带 token）

```
C→S {"type":"sub","symbols":["A_SHARE:600000","A_SHARE:000001"]}
C→S {"type":"unsub","symbols":[...]}
C→S {"type":"ping"}                        # 30s 心跳
S→C {"type":"pong"}
S→C {"type":"quote","data":[{symbol,last,pct,vol,ts},...]}   # 批量合并帧（权益差频 free 10s / vip 3s，B22；quote-push 合并窗口默认 200ms）
S→C {"type":"kline","data":{symbol,period,begin_ts,open,high,low,close,volume,amount}}  # 周期闭合 bar（B21 落地，字段对齐 proto Kline）
  # channels 订阅（B21 启用预埋字段，修订记录见 ADR-041）：
  # C→S {"type":"sub","symbols":["A_SHARE:600000"],"channels":["kline@1m"]}
  # 命名：<类型>@<period_min>，枚举 quote@3s(默认)/kline@1m/3m/5m/15m/30m/60m/1d（后六项数据源就绪后启用）
  # 权益（B22，docs/10 §3.4）：free 仅 kline@1m + quote 合并帧 10s；kline@3m+ 与 quote 3s 需 vip（vip1~3）——
  #   超档 channel 被 ws-gateway 忽略；权益数据走 Redis user_vip:{id}，5min 复核（ADR-040）
  # 兼容：缺省 channels = 纯 quote 行为，老客户端零感知
  # 闭合判定：floor(ts/period_ms) 窗口切换即闭合上一根（数据驱动）；亚分钟 3s/10s 仅实时流不落 CK
S→C {"type":"alert","data":{deliveryId,eventId,ruleId,symbol,title,detail,ts}}
 C→S {"type":"ack","deliveryId":"..."}
S→C {"type":"sys","data":{"code":"drain"}}  # 实例缩容通知，引导重连
```

约定：

1. 单连接订阅上限 200 symbol；帧速率上限 20 帧/s；
2. 告警消息客户端按 `deliveryId` 幂等去重，收到后发送 ACK（重连补拉 REST `/alerts?cursor=` 可能重发）；
3. 重连：1s 起指数退避（1/2/4/8...上限 60s）+ 随机抖动；重连成功先 REST 快照再订阅；
4. 页面不可见（H5）：`unsub` 取消行情订阅；告警仍经 alert-inbox 独立投递（不依赖 sub channels，勿发 `channels:["alert"]`——channels 枚举仅 quote@3s/kline@*）。
